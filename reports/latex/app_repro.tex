\begin{verbatim}
pip install -r requirements.txt
python src/01_generate_data.py          # sinh dữ liệu (42 kiểm tra nghiệm thu)
python src/02_data_understanding.py     # hiểu dữ liệu
python src/03_preprocess.py             # tiền xử lý (15 kiểm tra, đóng băng SHA-256)
python src/03b_evaluate_preprocessing.py  # chỉ chạy khi mã băm khớp

python src/05_cluster.py                # phân cụm
cd reports/latex && tectonic main.tex   # biên dịch báo cáo
\end{verbatim}}
